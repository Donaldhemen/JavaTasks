 // import scanner
// display enter  v0, v1, and t
// collect input for v0, v1, and t 
// declare average acceleration = (v1 - v0) / t
// display average acceleration

import java.util.Scanner;
public class AverageAcceleration {
    public static void main(String[] args) {
	Scanner input = new Scanner(System.in);

	System.out.println("Enter v0, v1, and t: ");
	double v0 = input.nextDouble(); 
	double v1 = input.nextDouble(); 
	double t = input.nextDouble(); 

	int averageAcceleration = (v1 - v0) / t;
	
	System.out.printf("The average acceleration is %d", averageAcceleration);
	}
}
